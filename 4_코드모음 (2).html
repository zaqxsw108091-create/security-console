# =====================================================================
# classifier.py  (4파트 대응·리포트 - 1번 파일)
#
# 이 파일이 하는 일
#   탐지 이벤트 1건의 위험 점수를 보고 "대응 등급" 글자 1개를 정한다.
#     "즉시 자동조치" : 사람 확인 없이 바로 막는다
#     "승인 후 조치"  : 사람이 승인하면 막는다
#     "모니터링만"    : 막지 않고 지켜본다
#   이 글자 3개는 팀 표준이다. 띄어쓰기까지 똑같이 써야 한다.
#   (1·3파트 코드와 2번 playbook.py가 이 글자를 그대로 비교한다. 한 글자라도 다르면 안 된다.)
#
# 이 파일이 "받아 오는 것" (값의 출처)
#   함수 안에서는 파일을 읽지 않는다. 값은 전부 def 괄호로 받는다.
#   받는 값의 원래 출처는 이렇다.
#     detections.json   : 2파트(탐지)가 만든 이벤트 목록 (공통로그 2번 형식)
#                         이 파일에서는 맨 아래 테스트에서만 읽는다.
#     vip_accounts.json : VIP 명단 파일. {"vip_users": [...], "vip_ips": [...]} 모양.
#                         이 파일에서는 맨 아래 테스트에서만 읽는다.
#   실제 서비스에서는 main.py가 두 파일을 읽어서 이 파일의 함수에 넣어 준다.
#
# 이 파일이 "내보내는 것"
#   event_check(...)가 돌려주는 등급 글자 1개
#   -> 2번 playbook.py가 `from classifier import event_check`로 가져가 쓴다.
#
# 주석 표시 약속
#   [뭐함] 이 줄(덩어리)이 하는 일
#   [출처] 이 이름의 값이 어디서 오는지
#   [흐름] 실제 값을 넣으면 어떻게 돌아가는지
#   [왜]   왜 필요한지
# =====================================================================

import json     # [뭐함] json 파일을 읽는 도구를 가져온다. 맨 아래 테스트의 json.load가 이 도구다.
                # [왜]   파일 맨 위에서 가져와야 아래 코드가 쓸 수 있다.


# ---------------------------------------------------------------------
# 1. check_vip : 이 계정(또는 IP)이 VIP인가?
# ---------------------------------------------------------------------
def check_vip(src_ip, user, vip_data):
    # [뭐함] VIP이면 True, 아니면 False를 돌려준다. (등급 글자는 여기서 정하지 않는다)
    # [출처] 괄호 안 세 이름은 이 함수를 부르는 쪽이 값을 넣어 준다.
    #        부르는 곳: 이 파일의 event_check 안 (아래 2번 함수)
    #          src_ip   : 이벤트의 src_ip 칸 (detections.json -> 공통로그 2번)
    #          user     : 이벤트의 user 칸    (detections.json -> 공통로그 2번)
    #          vip_data : vip_accounts.json을 읽어 만든 서랍장 (dict)
    #        주의: 괄호 순서가 (src_ip, user, vip_data)다. event_check의 순서(user, src_ip)와 다르다.
    # [흐름] check_vip("192.168.30.12", "emp3", vip_data)
    #        -> "emp3"는 vip_users에 없고 IP도 vip_ips에 없다 -> False

    if user in vip_data["vip_users"]:      # [뭐함] vip_data의 "vip_users" 서랍을 열어 user가 그 안에 있나 본다
        return True
    elif src_ip in vip_data["vip_ips"]:    # [뭐함] 아니면 "vip_ips" 서랍에 src_ip가 있나 본다
        return True
    else:                                  # [뭐함] 둘 다 아니면
        return False


# ---------------------------------------------------------------------
# 2. event_check : 대응 등급 정하기
# ---------------------------------------------------------------------
def event_check(user, src_ip, risk_score, rule, vip_data):
    # [뭐함] 이벤트 1건의 정보를 받아 등급 글자 1개를 돌려준다.
    # [출처] 괄호 안 다섯 이름의 값은 부르는 쪽이 넣어 준다.
    #        부르는 곳 1: 이 파일 맨 아래 테스트 for문 (event["user"] 등을 꺼내서 넣는다)
    #        부르는 곳 2: playbook.py의 run_playbook (event["user"] 등을 꺼내서 넣는다)
    #          user       : 이벤트의 user 칸       (2파트 공통로그 2번)
    #          src_ip     : 이벤트의 src_ip 칸     (2파트 공통로그 2번)
    #          risk_score : 이벤트의 risk_score 칸 (2파트가 계산한 위험 점수, 숫자)
    #          rule       : 이벤트의 rule 칸       (탐지 규칙 이름. 지금은 계산에 안 쓴다.
    #                       다른 파일과 칸을 맞추려고 받기만 한다)
    #          vip_data   : vip_accounts.json을 읽은 서랍장 (main.py가 읽어서 넘긴다)
    # [규칙] 점수 80 이상 -> 즉시 자동조치 / 50 이상 -> 승인 후 조치 / 나머지 -> 모니터링만
    #        단, 80 이상이어도 VIP면 승인 후 조치로 낮춘다.
    # [왜]   VIP 계정을 컴퓨터가 마음대로 막으면 큰 사고가 나므로 사람 승인을 거친다.

    # (1) 점수로 기본 등급을 level에 적는다.
    # [왜] 큰 점수부터 물어야 한다. if/elif는 위에서부터 처음 맞는 하나만 실행한다.
    #      90점을 ">= 50"부터 물으면 "승인 후 조치"에서 끝나 버린다.
    if risk_score >= 80:
        level = "즉시 자동조치"
    elif risk_score >= 50:
        level = "승인 후 조치"
    else:
        level = "모니터링만"

    # (2) 80점 이상인데 VIP이기까지 하면 level을 "승인 후 조치"로 고친다.
    # [출처] check_vip은 바로 위 1번 함수다. 괄호 순서 (src_ip, user, vip_data)를 맞춰 부른다.
    if risk_score >= 80 and check_vip(src_ip, user, vip_data) == True:
        level = "승인 후 조치"

    # (3) 마지막에 한 번만 돌려준다.
    # [왜] 위에서 바로 return 하면 거기서 함수가 끝나서 (2)번 VIP 검사가 실행되지 않는다.
    return level
    # [흐름] EVT-001(emp3, 점수 90): (1)에서 "즉시 자동조치" -> (2) VIP 아님 -> 그대로 돌려줌
    #        EVT-006(admin, 점수 90): (1)에서 "즉시 자동조치" -> (2) VIP -> "승인 후 조치"로 고침


# ---------------------------------------------------------------------
# 테스트용 (이 파일을 직접 실행할 때만 돈다)
# ---------------------------------------------------------------------
# [뭐함] __name__ 은 파이썬이 파일마다 자동으로 붙여 주는 이름이다.
#        이 파일을 직접 실행(VS Code의 ▶)하면 __name__ 이 "__main__" 이 되어 아래가 실행된다.
#        다른 파일이 `import classifier` 하면 __name__ 이 "classifier" 가 되어 아래는 실행되지 않는다.
# [왜]   이 조건이 없으면 두 가지 문제가 생긴다.
#        1) 다른 파일이 import 할 때마다 detections.json을 읽는다.
#           그 파일이 없는 폴더에서는 FileNotFoundError 로 프로그램이 멈춘다.
#        2) 테스트 for문의 출력("EVT-001 즉시 자동조치" 같은 줄)이 다른 파일의 실행 화면에 섞여 나온다.
if __name__ == "__main__":
    with open("detections.json", "r", encoding="utf-8") as file:   # [뭐함] 읽기("r")로 열고 file이라고 부른다
                                                 # [왜] encoding="utf-8" : 한글이 든 json을 윈도우에서 읽을 때 생기는
                                                 #      UnicodeDecodeError(cp949)를 막는다. 팀 파일은 utf-8로 저장한다.
        detections = json.load(file)             # [출처] 2파트가 만든 이벤트 목록(list)이 detections에 담긴다
                                                 # [흐름] 이벤트 7건이면 detections는 7칸짜리 목록

    with open("vip_accounts.json", "r", encoding="utf-8") as file: # [뭐함] 앞 with가 끝나서 file 이름을 다시 써도 된다
        vip_data = json.load(file)               # [출처] VIP 명단 서랍장(dict)이 vip_data에 담긴다
                                                 # [흐름] {"vip_users": ["admin"], "vip_ips": []}

    for event in detections:                     # [뭐함] 이벤트를 하나씩 꺼내 event라고 부른다
        result = event_check(event["user"], event["src_ip"], event["risk_score"],
                             event["rule"], vip_data)
                                                 # [출처] 괄호 안 순서는 event_check의 def 줄
                                                 #        (user, src_ip, risk_score, rule, vip_data)과 같다
        print(event["event_id"], result)         # 예: EVT-001 즉시 자동조치

# =====================================================================
# playbook.py  (4파트 대응·리포트 - 2번 파일)
#
# 이 파일이 하는 일 (세 가지)
#   1) 1번 classifier.py가 정한 대응 등급 글자를 받는다.
#   2) 이벤트가 지금 어느 단계인지(상태)를 statuses 서랍장에 적는다.
#   3) 1·3파트가 읽고 실행할 "조치 요청"(공통로그 3번, 6칸)을 만든다.
#
# 이 파일이 "받아 오는 것" (값의 출처)
#   classifier.py   : from classifier import event_check
#                     -> 등급 글자를 구하는 함수를 가져온다.
#                     (classifier.py는 import 해도 json 파일을 읽지 않는다)
#   event           : 2파트가 만든 detections.json의 이벤트 1건 (공통로그 2번).
#                     main.py가 목록에서 하나씩 꺼내 run_playbook에 넣어 준다.
#   vip_data        : vip_accounts.json을 읽은 서랍장. main.py가 읽어 넘긴다.
#   statuses        : {이벤트번호: 상태} 서랍장. main.py가 빈 {}로 만들어 계속 넘겨 준다.
#   result          : 1·3파트가 조치 실행 뒤 돌려주는 조치 결과 (공통로그 5번).
#
# 이 파일이 "내보내는 것"
#   run_playbook(...)   -> 조치 요청 서랍장 6칸 (또는 None). 1·3파트가 읽는다.
#                          또 statuses에 이 이벤트의 처음 상태를 적는다.
#   apply_result(...)   -> 돌려주는 값은 없고 statuses의 상태를 바꾼다.
#   (조치 요청은 3번 false_positive.py와 4번 ticket.py의 재료이기도 하다)
#
# 팀 표준 글자 (띄어쓰기까지 그대로)
#   등급 : 즉시 자동조치 / 승인 후 조치 / 모니터링만
#   상태 : 신규 / 승인대기 / 조치중 / 완료 / 실패 / 모니터링
#   결과 : success / fail / skipped / approval_required
#
# 주석 표시 약속
#   [뭐함] 이 줄(덩어리)이 하는 일   [출처] 값이 어디서 오는지
#   [흐름] 실제 값을 넣으면 어떻게 돌아가는지   [왜] 왜 필요한지
# =====================================================================

import json
from classifier import event_check
# [뭐함] classifier.py 파일 안의 event_check 함수를 이 파일로 가져온다.
# [왜]   등급 정하는 일은 1번 파일이 이미 한다. 다시 쓰지 않고 가져다 쓴다.
# [주의] classifier.py가 이 파일과 같은 폴더에 있어야 한다.
# [참고] import json 은 맨 아래 테스트에서는 쓰지 않는다. 지워도 동작은 같다.


# ---------------------------------------------------------------------
# 1. choose_action : 이벤트 유형 -> 조치(action) 이름
# ---------------------------------------------------------------------
def choose_action(event_type):
    # [뭐함] 이벤트 유형 글자를 받아 조치 이름 글자를 돌려준다. 모르는 유형이면 None.
    # [출처] event_type : 부르는 곳은 아래 3번 make_request.
    #        값은 이벤트의 event_type 칸 (2파트 공통로그 2번)
    #        ABNORMAL_LOGIN / OFFHOUR_LOGIN / UNAUTHORIZED_ACCESS /
    #        MALICIOUS_IP / TRAFFIC_SPIKE / CORRELATED_ATTACK 중 하나
    # [왜]   조치를 실행하는 1·3파트는 "무슨 조치인지" 글자를 보고 움직이므로 우리가 정해서 넘긴다.
    # [흐름] "OFFHOUR_LOGIN" -> 두 번째 elif에 맞아 "terminate_session" 돌려주고 끝

    if event_type in ["ABNORMAL_LOGIN", "TRAFFIC_SPIKE", "CORRELATED_ATTACK", "MALICIOUS_IP"]:
        return("block_ip")             # IP 차단 : 3파트가 실행한다
    elif event_type == "OFFHOUR_LOGIN":
        return("terminate_session")    # 세션 종료 : 1파트가 실행한다
    elif event_type == "UNAUTHORIZED_ACCESS":
        return("lock_account")         # 계정 잠금 : 1파트가 실행한다
    else:
        return None                    # 규칙에 없는 유형. 아래 run_playbook이 None을 보고 요청을 만들지 않는다


# ---------------------------------------------------------------------
# 2. choose_target : 조치(action) -> 조치 대상(target)
# ---------------------------------------------------------------------
def choose_target(action, user, src_ip):
    # [뭐함] 조치 이름에 맞는 대상(IP 또는 계정)을 돌려준다.
    # [출처] action : 바로 위 1번 함수가 돌려준 값 (make_request 안에서 변수 action에 담긴 것)
    #        user, src_ip : 이벤트의 user, src_ip 칸 (2파트 공통로그 2번)
    # [왜]   IP를 막는 조치는 IP가, 계정에 하는 조치는 계정 이름이 필요하다.
    # [주의] 값이 필요하므로 return(src_ip)에 따옴표를 쓰지 않는다. 따옴표를 쓰면 "src_ip"라는 글자가 간다.
    # [흐름] action="block_ip", src_ip="192.168.30.12" -> "192.168.30.12" 돌려준다

    if action == "block_ip":
        return(src_ip)
    elif action in ["lock_account", "terminate_session"]:
        return(user)
    else:
        return None                    # action이 None이면 target도 None


# ---------------------------------------------------------------------
# 3. make_request : 조치 요청(공통로그 3번) 1건 만들기
# ---------------------------------------------------------------------
def make_request(event, level, dry_run):
    # [뭐함] 1·3파트에게 넘길 조치 요청 서랍장(6칸)을 만들어 돌려준다.
    # [출처] event   : 이벤트 1건. run_playbook(아래 6번)이 그대로 넘겨 준다. (2파트 공통로그 2번)
    #        level   : 대응 등급 글자. run_playbook이 classifier.event_check로 구한 값이다.
    #        dry_run : True면 "기록만, 실제 조치는 안 함", False면 진짜 실행.
    #                  run_playbook이 False를 넣어 준다. 입력에도 없고 만들 수도 없어서 괄호로 받는다.
    # [왜]   칸 이름 6개(event_id, action, target, response_level, reason, dry_run)는
    #        공통로그 3번이 정한 팀 표준이다. 1·3파트가 이 이름표로 값을 꺼내므로 바꾸면 안 된다.

    action = choose_action(event["event_type"])                      # 위 1번 함수에게 물어서 action에 담는다
    target = choose_target(action, event["user"], event["src_ip"])   # 위 2번 함수. 괄호 순서는 def (action, user, src_ip)와 같다

    request = {
        "event_id"       : event["event_id"],    # [출처] 이벤트에 있는 값을 그대로
        "action"         : action,               # [출처] 바로 위에서 구한 변수
        "target"         : target,               # [출처] 바로 위에서 구한 변수
        "response_level" : level,                # [출처] def 괄호로 받은 등급 글자
        "reason"         : event["rule"],        # [출처] 이벤트의 rule 칸(탐지 규칙 이름)을 이유로 쓴다
        "dry_run"        : dry_run               # [출처] def 괄호로 받은 값
    }
    return request                               # [주의] 함수 이름이 아니라 서랍장이 든 변수 request를 돌려준다
    # [흐름] event EVT-001(ABNORMAL_LOGIN, emp3, 192.168.30.12, rule=failed_login_5_in_10min),
    #        level="즉시 자동조치", dry_run=False 이면
    #        {"event_id":"EVT-001","action":"block_ip","target":"192.168.30.12",
    #         "response_level":"즉시 자동조치","reason":"failed_login_5_in_10min","dry_run":False}


# ---------------------------------------------------------------------
# 4. first_status : 대응 등급 -> 이벤트의 처음 상태
# ---------------------------------------------------------------------
def first_status(level):
    # [뭐함] 등급 글자를 받아 처음 상태 글자를 돌려준다.
    # [출처] level : run_playbook(아래 6번)이 event_check로 구해서 넘긴 등급 글자.
    # [주의] 점수(숫자)가 아니라 등급 글자를 비교한다.
    #        등급 글자는 "모니터링만", 상태 글자는 "모니터링"이다. 비교할 때는 등급 쪽을 쓴다.
    # [흐름] "승인 후 조치" -> "승인대기"

    if level == "즉시 자동조치":
        return("조치중")
    elif level == "승인 후 조치":
        return("승인대기")
    elif level == "모니터링만":
        return("모니터링")
    else:
        return("신규")                  # 정해진 등급이 아니면 신규

# 5. apply_result : 조치 결과(공통로그 5번) -> 상태 바꾸기
# ---------------------------------------------------------------------
def apply_result(statuses, result):
    # [뭐함] 1·3파트가 돌려준 조치 결과를 보고 statuses의 상태를 바꾼다. return이 없다.
    # [출처] statuses : main.py가 만든 상태 서랍장. run_playbook이 이미 이 이벤트 칸을 만들어 두었다.
    #        result   : 1·3파트가 돌려준 조치 결과 1건 (공통로그 5번).
    #                   쓰는 칸은 두 개 : result["event_id"], result["result"]
    # [왜]   return이 없는 이유: 서랍장은 함수 안에서 바꿔도 바깥 서랍장이 같이 바뀐다.
    #        그래서 값을 돌려줄 필요가 없다.
    # [흐름] result = {"event_id":"EVT-003","result":"fail", ...}
    #        -> event_id="EVT-003", outcome="fail" -> statuses["EVT-003"] = "실패"

    event_id = result["event_id"]

    outcome = result["result"]         # success / fail / skipped / approval_required 중 하나

    if outcome == "success":
        statuses[event_id] = "완료"
    elif outcome == "fail":
        statuses[event_id] = "실패"
    elif outcome == "skipped":
        statuses[event_id] = "완료"        # 이미 처리되어 건너뜀 -> 일은 끝난 것으로 본다
        # [팀이 정할 것] ticket.py의 update_ticket은 skipped일 때 티켓 상태를 바꾸지 않는다(조치중 그대로).
        #        같은 이벤트인데 이벤트 상태(완료)와 티켓 상태가 다르다. 팀이 정하면 한쪽을 고친다.
        #        고치는 법은 5_상황별_고치기 6장.
    elif outcome == "approval_required":
        statuses[event_id] = "승인대기"

            statuses[event_id] = "신규"
            return None
        return request
    # [흐름] EVT-004(점수 낮음) -> level="모니터링만" -> statuses["EVT-004"]="모니터링" -> None

# ---------------------------------------------------------------------
# 함수 1. choose_restore_action
# ---------------------------------------------------------------------
def choose_restore_action(action):
    # [뭐함] 원래 조치 글자를 받아서, 그걸 되돌리는 조치 글자를 돌려준다.
    # [값의 출처] action : def 괄호. 부르는 곳은 아래 make_restore_request이고,
    #            값은 원래 조치 요청의 action 칸(request["action"])이다.
    # [왜]   "무엇의 반대가 무엇인지"를 한 곳에만 적어 두면 틀릴 일이 없다.
    # [흐름] action = "block_ip"           -> "unblock_ip" 돌려주고 끝
    #        action = "lock_account"       -> "unlock_account" 돌려주고 끝
    #        action = "terminate_session"  -> 위 둘이 아니므로 None 돌려주고 끝

    if action == "block_ip":                 # [뭐함] 원래 조치가 IP 차단이면
        return("unblock_ip")                 #        IP 차단 해제로 되돌린다
    elif action == "lock_account":           # [뭐함] 아니고, 계정 잠금이면
        return("unlock_account")             #        계정 잠금 해제로 되돌린다
    else:                                    # [뭐함] 둘 다 아니면
        return None                          # [왜] terminate_session은 이미 끊은 접속을
                                             #      되돌릴 방법이 없다. None은 "되돌릴 게 없다"는 표시다.


# ---------------------------------------------------------------------
# 함수 2. make_restore_request
# ---------------------------------------------------------------------
def make_restore_request(request, result):
    # [뭐함] 복구 요청 서랍장(6칸)을 만들어 돌려준다. 복구할 게 없으면 None을 돌려준다.
    # [값의 출처]
    #   request : playbook.py가 만든 원래 조치 요청 (칸 6개: event_id, action, target,
    #             response_level, reason, dry_run)
    #   result  : 1·3파트가 돌려준 조치 결과 (그 안의 "result" 칸이 success/fail 등의 글자)
    # [왜]   복구 요청은 "원래 조치가 실제로 성공했을 때만" 의미가 있다.
    #        안 걸린 조치를 풀려고 하면 오히려 오류가 난다.
    # [흐름] (예) request의 action이 "block_ip", result의 "result"가 "success"
    #        -> 3개의 if를 모두 통과 -> unblock_ip 복구 요청을 돌려준다.

    # [뭐함] 되돌리는 조치 글자를 함수 1에게 받아서 restore_action에 담는다.
    # [흐름] request["action"]이 "block_ip"이면 restore_action에는 "unblock_ip"가 담긴다.
    # [왜]   글자를 직접 적지 않고 함수에게 받는 이유: 요청마다 action이 다르기 때문이다.
    restore_action = choose_restore_action(request["action"])

    # [뭐함] 조치 결과가 success가 아니면 복구하지 않는다.
    # [왜]   fail            : 조치가 걸리지 않았으니 되돌릴 게 없다.
    #        skipped         : 실행하지 않았으니 되돌릴 게 없다.
    #        approval_required: 아직 승인 대기라 실행 전이다.
    if result["result"] != "success":
        return None

    # [뭐함] 원래 요청이 dry_run(연습 실행)이었으면 복구하지 않는다.
    # [왜]   dry_run이 True면 실제 조치 없이 기록만 했으므로 되돌릴 것이 없다.
    if request["dry_run"] == True:
        return None

    # [뭐함] 되돌리는 조치가 없으면(None) 복구하지 않는다.
    # [왜]   terminate_session 같은 조치는 함수 1이 None을 돌려준다.
    if restore_action == None:
        return None

    # [뭐함] 세 검사를 모두 통과했으니 복구 요청 서랍장 6칸을 만든다.
    # [왜]   1·3파트가 읽는 조치 요청 형식(6칸)과 똑같이 만들어야 그대로 실행할 수 있다.
    data = {
        "event_id": request["event_id"],       # 원래 요청과 같은 이벤트라서 그대로 옮긴다
        "action": restore_action,              # 바뀐 칸 1 : 되돌리는 조치 글자
        "target": request["target"],           # 같은 대상(IP 또는 계정)을 풀어야 하므로 그대로 옮긴다
        "response_level": "즉시 자동조치",       # 팀 표준 글자. 띄어쓰기까지 정확히 이대로
        "reason": "false_positive",            # 바뀐 칸 2 : 오탐 때문에 되돌린다는 이유
        "dry_run": request["dry_run"]          # 위 검사를 통과했으니 항상 False다
    }

    return(data)                               # [뭐함] 완성한 복구 요청을 돌려준다

# ---------------------------------------------------------------------

def save_feedback(feedback, filename):
    # [뭐함] 기록 한 건(feedback)을 json 파일의 목록 맨 뒤에 추가한다.
    # [값의 출처] feedback : 함수 3이 만든 서랍장 / filename : 파일 이름 글자
    # [왜]   파일에는 예전 기록이 쌓여 있다. 한 건만 새로 적으면 예전 기록이 사라진다.
    #        그래서 "읽기 -> 추가 -> 다시 쓰기" 순서로 한다.
    # [흐름] 처음(파일 없음)  : records = [] -> 한 건 추가 -> 1건짜리 파일이 생긴다.
    #        두 번째(1건 있음): records = [1건] -> 한 건 추가 -> 2건짜리 파일로 다시 쓴다.
    # 이 함수는 파일에 적기만 하므로 return이 없다.

    if os.path.exists(filename):                 # [뭐함] 파일이 있는지 먼저 확인한다.
                                                 # [왜]   처음에는 파일이 없다. 없는 파일을
                                                 #        "r"로 열면 에러가 나기 때문이다.
        with open(filename, "r", encoding="utf-8") as file:        # [뭐함] 읽기("r")로 연다
                                                 # [왜] encoding="utf-8" : 쓸 때와 읽을 때 방식을 같게 하고,
                                                 #      다른 파트(2파트)가 utf-8로 읽어도 한글이 안 깨지게 한다.
                                                 # [주의] 예전에 encoding 없이 만든 false_positive_feedback.json이
                                                 #        있으면 읽다가 UnicodeDecodeError가 난다. 그 파일을 지우고 다시 실행한다.
            records = json.load(file)            # [뭐함] 안에 있는 목록을 records에 담는다
    else:
        records = []                             # [뭐함] 파일이 없으면 빈 목록에서 시작한다

    records.append(feedback)                     # [뭐함] 목록 맨 뒤에 새 기록 한 건을 추가한다

    with open(filename, "w", encoding="utf-8") as file:            # [뭐함] 쓰기("w")로 연다
                                                 # [왜]   "w"는 열면서 기존 내용을 전부 지운다.
                                                 #        앞에서 예전 기록을 records에 담아 둔 이유다.
        json.dump(records, file, ensure_ascii=False, indent=4)
                                                 # [뭐함] records 전체를 파일에 적는다
                                                 # ensure_ascii=False : 한글이 \uXXXX로 깨지지 않게 한다
                                                 # indent=4           : 사람이 읽기 좋게 줄을 맞춘다

# 이 파일이 "내보내는 것"
#   make_ticket(...)        -> 티켓 서랍장 (아래 칸 목록 참고). 5번 report.py가 이 티켓으로 리포트를 만든다.
#   update_ticket(...)      -> 갱신된 티켓 (같은 서랍장이 바뀐 것)

#                    (event_id는 "event" 서랍에 이미 있어서 여기서는 뺐다)

#
# 팀 표준 글자

# ---------------------------------------------------------------------
# 1. make_ticket : 티켓 1장 만들기
# ---------------------------------------------------------------------
def make_ticket(event, request, status):
    # [뭐함] 이벤트 1건과 그 조치 요청, 처음 상태로 티켓 서랍장 한 장을 만들어 돌려준다.
    # [출처] event   : def 괄호. 공통로그 2번의 이벤트 1건. (11칸이 모두 들어 있어야 한다)
    #        request : def 괄호. 공통로그 3번의 조치 요청 1건. (run_playbook이 돌려준 것)
    #        status  : def 괄호. playbook.py의 statuses[event_id] 값. 예) "조치중", "승인대기"
    # [왜]   status를 이 함수 안에서 정하지 않고 괄호로 받는 이유:
    #        상태를 정하는 곳을 playbook.py 한 군데로 두기 위해서다.
    #        여기서 또 정하면 두 파일의 상태가 어긋날 수 있다.
    # [주의] 모니터링만이면 run_playbook이 request로 None을 돌려준다.
    #        이 함수는 None을 받으면 request["action"]에서 에러가 난다.
    #        그래서 main.py에서 request가 None이 아닐 때만 이 함수를 불러야 한다.
    # [흐름] event_id가 "EVT-001"이면 ticket_id는 "TKT-EVT-001"이 된다.

    ticket_id = "TKT-" + event["event_id"]     # [뭐함] 글자 "TKT-" 뒤에 이벤트 번호를 붙여 티켓 번호를 만든다

    ticket = {
        "event": {                                          # 이벤트 정보 서랍 (공통로그 2번 11칸)
            "event_id": event["event_id"],                  # [출처] event 서랍장의 같은 이름 칸
            "event_type": event["event_type"],
            "user": event["user"],
            "src_ip": event["src_ip"],
            "risk_score": event["risk_score"],
            "priority": event["priority"],
            "asset": event["asset"],
            "impact_scope": event["impact_scope"],          # [주의] 철자 impact_scope (impct, imapct 아님)
            "detected_at": event["detected_at"],
            "rule": event["rule"],                          # 리포트의 "탐지 근거"에 쓴다
            "evidence": event["evidence"]                   # 리포트의 "탐지 근거"에 쓴다
                                                            # [주의] 공통로그 2번 카드에서 evidence는 글자가 아니라 "목록"이다.
                                                            #        예: ["2026-11-02 02:10:11", "2026-11-02 02:14:05"]
                                                            #        여기서는 목록을 그대로 옮기기만 한다. 글자로 바꾸는 일은 report.py가 한다.
        },
        "request": {                                        # 조치 요청 서랍 (공통로그 3번 5칸)
            "action": request["action"],                    # [출처] request 서랍장의 같은 이름 칸
            "target": request["target"],
            "response_level": request["response_level"],
            "reason": request["reason"],
            "dry_run": request["dry_run"]
        },
        "ticket_id": ticket_id,                             # [출처] 위에서 만든 변수
        "status": status,                                   # [출처] def 괄호로 받은 상태 글자
        "action_result": None,                              # 조치 결과 글자. 결과가 오기 전이라 None. update_ticket이 채운다
        "message": None,                                    # 조치 결과 설명 글자. update_ticket이 채운다
        "updated_at": event["detected_at"]                  # 처음에는 탐지 시각. update_ticket이 결과 시각으로 바꾼다
    }

    return(ticket)                                          # [주의] 돌려주는 줄이 없으면 호출한 쪽에는 None이 간다


# ---------------------------------------------------------------------
# 2. update_ticket : 조치 결과로 티켓 갱신하기
# ---------------------------------------------------------------------
def update_ticket(ticket, result):
    # [뭐함] 조치 결과(공통로그 5번)를 받아 티켓의 결과 칸, 설명 칸, 시각 칸, 상태 칸을 바꾼다.
    # [출처] ticket : def 괄호. make_ticket이 만든 티켓 서랍장.
    #        result : def 괄호. 1·3파트가 돌려준 조치 결과 1건. 쓰는 칸은 result, message, timestamp 셋.
    # [왜]   조치 결과 글자(result["result"])에 따라 상태를 다르게 정한다.
    # [팀이 정할 것] 강의 자료는 "조치가 성공했다고 티켓을 자동으로 닫지 말라"고 한다.
    #        지금 코드는 success이면 "완료"로 바꾼다. 팀 표준 상태 글자 6개에
    #        "사람이 확인하는 단계"가 없어서 이렇게 정했다. 팀이 다르게 정하면 이 줄만 고친다.
    # [규칙] success           -> "완료"
    #        fail              -> "실패"
    #        approval_required -> "승인대기"
    #        skipped           -> 상태는 그대로 두고 결과·설명·시각만 적는다.
    # [팀이 정할 것] skipped일 때 playbook.py의 apply_result는 이벤트 상태를 "완료"로 바꾼다.
    #        여기서는 티켓 상태를 바꾸지 않는다(조치중 그대로). 같은 이벤트인데 두 상태가 다르다.
    #        팀이 정하면 한쪽을 고친다. 고치는 법은 5_상황별_고치기 6장.
    # [흐름] result["result"]가 "success"이고 timestamp가 "2026-11-02 02:15:05"이면
    #        action_result="success", updated_at="2026-11-02 02:15:05", status="완료"

    ticket["action_result"] = result["result"]       # [뭐함] 결과 글자를 티켓에 적는다
    ticket["message"] = result["message"]            # [뭐함] 결과 설명 글자를 티켓에 적는다
    ticket["updated_at"] = result["timestamp"]       # [뭐함] 갱신 시각을 결과가 만들어진 시각으로 바꾼다

    if result["result"] == "success":                # [뭐함] 결과 글자에 따라 상태를 정한다
        ticket["status"] = "완료"
    elif result["result"] == "fail":
        ticket["status"] = "실패"
    elif result["result"] == "approval_required":
        ticket["status"] = "승인대기"
    else:

def read_log(filename):
    # [뭐함] 파일 이름 글자를 받아, 그 파일 안의 조치 이력 목록을 돌려준다.
    # [출처] filename : def 괄호. 6번 main.py가 "containment_log.json"을 넣어 준다.
    # [흐름] filename이 "containment_log.json"이면 파일 안의 [ {...}, {...} ] 목록이 logs에 담겨 돌아온다.

    with open(filename, "r", encoding="utf-8") as file:   # [뭐함] 읽기("r")로 열고 file이라고 부른다
                                                          # [왜] encoding="utf-8"이 없으면 윈도우에서
                                                          #      한글이 든 파일을 읽다가 에러가 날 수 있다.
        logs = json.load(file)                            # [뭐함] 파일 안의 글자를 파이썬 목록으로 바꿔 logs에 담는다
    return (logs)                                         # [주의] with와 같은 칸(4칸)에 쓴다. with 밖에서 돌려준다.

def make_report(ticket, logs):

def save_report(lines, filename):
